```bash
"C:/Users/lizhicai/.workbuddy-ai/binaries/node/versions/22.22.2-2/node.exe" \
  "C:/Users/lizhicai/.workbuddy-ai/binaries/node/versions/22.22.2-2/node_modules/corepack/dist/corepack.js" \
  pnpm dev
```
